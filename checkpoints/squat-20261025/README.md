## 태그에서 안전하게 꺼내기

```bash

mkdir -p /tmp/squat-tag-export

git archive squat-verified-20261025 checkpoints/squat-20261025 \
  | tar -x -C /tmp/squat-tag-export

python3 /tmp/squat-tag-export/checkpoints/squat-20261025/restore.py \
  --destination /tmp/squat-restored-workspace

  --verify-only /tmp/squat-restored-workspace
```
